/* Operation Acorn: gear and kits (the gear package).
   The research team backpack, waist bag and fuel can, the welding mask, the four modification kits and the three
   weapon kits. Every property is one the vanilla scripts use: the backpack is modelled on Bag_ALICEpack, the waist bag on
   Bag_FannyPackFront/Back (media/scripts/generated/items/container.txt) and the can on PetrolCan (normal.txt). The team's
   welder is the vanilla blowtorch with the team's mark in its modData (Gear/Gear.lua): the game checks the torch by its exact type
   (barricading, unbarricading, scrapping metal, burnt cars), so one of another type would not work. The field tools are
   in OA_field_tools.txt.
   What scripts cannot do is in media/lua/shared/OperationAcorn/Gear: fitting a kit (a timed action that writes the
   counts to the item, the bag or the vehicle) and redeeming the handover voucher. The tooltips are set in code
   (ISToolTipInv, Gear/Menu.lua), so none is named here. */
module OperationAcorn
{
    imports
    {
        Base
    }

    /* The research team backpack: 35 capacity and 85 weight reduction, the vanilla maxima. Two military canteens
       are built into its side pouches (they are part of the model): together they are the pack's water storage, the
       FluidContainer below, as much as the two canteens (CanteenMilitary, 0.9 each) and empty when bought. Like the
       canteen it takes any fluid.
       WebbingLeft/Right are the hotbar hangers the game places on the shoulder straps at the chest (a radio, a knife;
       the ALICE suspenders have the same two), where this pack's straps run, so they stay; they are clear of the
       canteens on the sides. */
    item TeamBackpack
    {
        DisplayCategory = Bag,
        ItemType = base:container,
        Weight = 2.0,
        Icon = OA_ResearchBackpack,
        AttachmentReplacement = Bag,
        BloodLocation = Bag,
        CanBeEquipped = base:back,
        CanHaveHoles = false,
        Capacity = 35,
        CloseSound = CloseBag,
        ClothingItem = OA_Bag_ResearchPack,
        EquipSound = EquipBackpackLarge,
        OpenSound = OpenBag,
        PutInSound = PutItemInBag,
        ReplaceInPrimaryHand = OA_Bag_ResearchPack_RHand holdingbagright,
        ReplaceInSecondHand = OA_Bag_ResearchPack_LHand holdingbagleft,
        RunSpeedModifier = 0.94,
        WeightReduction = 85,
        WorldStaticModel = OperationAcorn.OA_ResearchPack_Ground,
        AttachmentsProvided = WebbingLeft;WebbingRight,
        Tags = base:ignorezombiedensity;base:omitemptyfromname,
        component FluidContainer
        {
            ContainerName = OA_WaterStorage,
            Capacity = 1.8,
            CustomDrinkSound = DrinkingFromBottlePlastic,
        }
    }

    /* The research team waist bag: worn at the front or, by the same menu as the vanilla fanny pack's, at the back
       (ClothingItemExtra: the game makes the other item and moves the contents and the modData across). More than a
       fanny pack holds (1) and less than a satchel; the weight reduction is the vanilla bags' most. Its own models
       (tools/waistbag_mesh.py, scripts/OA_waistbag_models.txt). The kits fit it as they fit a backpack. */
    item TeamWaistBagFront
    {
        DisplayCategory = Bag,
        ItemType = base:container,
        Weight = 0.5,
        Icon = OA_ResearchWaistBag,
        BodyLocation = base:fannypackfront,
        CanBeEquipped = base:fannypackfront,
        CanHaveHoles = false,
        Capacity = 6,
        CloseSound = CloseBag,
        ClothingItem = OA_Bag_ResearchWaistFront,
        ClothingItemExtra = OperationAcorn.TeamWaistBagBack,
        ClothingItemExtraOption = FannyPack_WearBack,
        EquipSound = EquipFannyPack,
        OpenSound = OpenBag,
        PutInSound = PutItemInBag,
        RunSpeedModifier = 0.98,
        WeightReduction = 85,
        ClothingExtraSubmenu = FannyPack_WearFront,
        WorldStaticModel = OperationAcorn.OA_ResearchWaist_Ground,
        Tags = base:ignorezombiedensity,
    }

    item TeamWaistBagBack
    {
        DisplayCategory = Bag,
        ItemType = base:container,
        Weight = 0.5,
        Icon = OA_ResearchWaistBag,
        BodyLocation = base:fannypackback,
        CanBeEquipped = base:fannypackback,
        CanHaveHoles = false,
        Capacity = 6,
        CloseSound = CloseBag,
        ClothingItem = OA_Bag_ResearchWaistBack,
        ClothingItemExtra = OperationAcorn.TeamWaistBagFront,
        ClothingItemExtraOption = FannyPack_WearFront,
        EquipSound = EquipFannyPack,
        OpenSound = OpenBag,
        PutInSound = PutItemInBag,
        RunSpeedModifier = 0.98,
        WeightReduction = 85,
        ClothingExtraSubmenu = FannyPack_WearBack,
        WorldStaticModel = OperationAcorn.OA_ResearchWaist_Ground,
        Tags = base:ignorezombiedensity,
    }

    /* The research team fuel can: 12 L, a little more than the vanilla petrol can's 10, at its empty weight. The kits
       make it hold more and weigh less (Gear/Gear.lua). The shop sends it empty. The base:petrol tag is what the
       vehicle and generator menus look for. Its own jerry can model (tools/fuelcan_mesh.py,
       scripts/OA_fuelcan_models.txt); the hand holds it where it holds the vanilla can. */
    item TeamFuelCan
    {
        DisplayCategory = VehicleMaintenance,
        Weight = 1.6,
        ItemType = base:normal,
        Icon = OA_TeamFuelCan,
        StaticModel = OperationAcorn.OA_TeamFuelCan,
        ReplaceInSecondHand = OA_Bag_TeamFuelCan_LHand holdingbagleft,
        ReplaceInPrimaryHand = OA_Bag_TeamFuelCan_RHand holdingbagright,
        WorldStaticModel = OperationAcorn.OA_TeamFuelCan_Ground,
        Tags = base:petrol;base:cookablemicrowave,
        component FluidContainer
        {
            ContainerName = GasCan,
            Capacity = 12.0,
            Fluids
            {
                fluid = Petrol:1.0,
            }
        }
    }

    /* The modification kits, one per thing they improve (NUMBERS 7): the expansion kit (what a bag, the fuel can, the
       welder or a vehicle's storage holds), the lightening kit (what a bag or a team item weighs), the reinforcement
       kit (how long a field tool lasts) and the engine tuning kit. Each is fitted by a timed action
       (Gear/Actions.lua). */
    item KitCapacity
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_CapacityKit,
        WorldStaticModel = Base.ToolBox_Ground,
    }

    item KitLightening
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_LighteningKit,
        WorldStaticModel = Base.SewingBox,
    }

    item KitReinforcement
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_ReinforcementKit,
        WorldStaticModel = Base.SewingBox,
    }

    item KitEngine
    {
        DisplayCategory = VehicleMaintenance,
        ItemType = base:normal,
        Weight = 2.0,
        Icon = OA_EngineTuningKit,
        WorldStaticModel = Base.EngineParts,
    }

    /* The weapon kits (NUMBERS 7 'Weapon kits'): fitted to a melee weapon or a firearm by the same timed action as the
       others. The damage kit raises its least and most damage, the critical kit its critical chance, the repair kit
       puts condition back. They have the modification kits' icons until their own are drawn. */
    item KitWeaponDamage
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_ReinforcementKit,
        WorldStaticModel = Base.SewingBox,
    }

    item KitWeaponCrit
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_LighteningKit,
        WorldStaticModel = Base.SewingBox,
    }

    item KitWeaponRepair
    {
        DisplayCategory = Tool,
        ItemType = base:normal,
        Weight = 1.0,
        Icon = OA_CapacityKit,
        WorldStaticModel = Base.ToolBox_Ground,
    }

    /* The team welding mask: the vanilla WeldingMask (clothing.txt) with a vented shell, so it is cooler and less in
       the way: Insulation 0.25 -> 0.1, WindResistance 0.55 -> 0.3, DiscomfortModifier 0.2 -> 0.05. The vision through
       the lens is the vanilla mask's (0.5), as is the protection. The base:weldingmask tag is what welding looks for.
       Its model is the vanilla mask's in the team's olive (tools/vanilla_art.py). */
    item TeamWeldingMask
    {
        DisplayCategory = Tool,
        ItemType = base:clothing,
        Icon = OA_TeamWeldingMask,
        BiteDefense = 50,
        BloodLocation = Head,
        BodyLocation = base:maskfull,
        CanHaveHoles = false,
        ClothingItem = OA_Hat_TeamWelderMask,
        Insulation = 0.1,
        ScratchDefense = 70,
        WaterResistance = 1.0,
        WindResistance = 0.3,
        Tags = base:weldingmask;base:isdisguise,
        VisionModifier = 0.5,
        DiscomfortModifier = 0.05,
    }
}
