/* Operation Acorn: gear and kits (the gear package).
   The research team backpack, the improved fuel cans and welding torches (Mk.I to Mk.III) and the four modification
   kits. Every property is one the vanilla scripts use: the backpack is modelled on Bag_ALICEpack, the cans on
   PetrolCan (media/scripts/generated/items/normal.txt) and the torches on BlowTorch (drainable.txt).
   What scripts cannot do is in media/lua/shared/OperationAcorn/Gear: fitting a kit (a timed action that writes the
   counts to the bag or the vehicle), redeeming the handover voucher, and teaching the vanilla welding recipes about
   the Mk torches. The tooltips are set in code (ISToolTipInv, Gear/Menu.lua), so none is named here. */
module OperationAcorn
{
    imports
    {
        Base
    }

    /* The research team backpack: 35 capacity and 85 weight reduction, the vanilla maxima, with two webbing
       hangers on the sides (WebbingLeft/Right) for the two military canteens the shop gives with it. */
    item TeamBackpack
    {
        DisplayCategory = Bag,
        ItemType = base:container,
        Weight = 2.0,
        Icon = OA_ResearchBackpack,
        AttachmentReplacement = Bag,
        BloodLocation = Bag,
        CanBeEquipped = base:back,
        CanHaveHoles = false,
        Capacity = 35,
        CloseSound = CloseBag,
        ClothingItem = OA_Bag_ResearchPack,
        EquipSound = EquipBackpackLarge,
        OpenSound = OpenBag,
        PutInSound = PutItemInBag,
        ReplaceInPrimaryHand = OA_Bag_ResearchPack_RHand holdingbagright,
        ReplaceInSecondHand = OA_Bag_ResearchPack_LHand holdingbagleft,
        RunSpeedModifier = 0.94,
        WeightReduction = 85,
        WorldStaticModel = OperationAcorn.OA_ResearchPack_Ground,
        AttachmentsProvided = WebbingLeft;WebbingRight,
        Tags = base:ignorezombiedensity,
    }

    /* The improved welding torches: the vanilla BlowTorch (10 uses, 1.0 full and 0.3 empty) with the charge times
       1.5, 2 and 3 (UseDelta is 1 / the uses; a hair under, so that the game's floor(1 / UseDelta) gives 15, 20 and
       30) and the weight times 1, 0.6 and 0.4 (both the full and the empty weight). base:blowtorch is the tag the
       vanilla code that is not tied to the exact item type looks for. */
    item TorchMk1
    {
        DisplayCategory = Tool,
        ItemType = base:drainable,
        Weight = 1.0,
        WeightEmpty = 0.3,
        Icon = OA_WeldingTorchMk1,

        UseWhileEquipped = false,
        cantBeConsolided = true,

    /* Refilling an improved torch from a propane tank: the vanilla RefillBlowTorch recipe (recipes_metalWelding.txt)
       for each Mk. The torch is replaced by a new one of its own kind, filled by the same code (the propane that
       fills one tank-load of the torch's own uses). */
    craftRecipe RefillTorchMk1
    {
        timedAction = Welding,
        time = 50,
        OnCreate = RecipeCodeOnCreate.refillBlowTorch,
        Tags = InHandCraft;Welding,
        category = Metalworking,
        inputs
        {
            item 1 [OperationAcorn.TorchMk1] mode:destroy flags[NotFull;ItemCount],
            item 1 [Base.PropaneTank] flags[NotEmpty;ItemCount] mode:keep,
        }
        outputs
        {
            item 1 OperationAcorn.TorchMk1,
        }
    }

    craftRecipe RefillTorchMk2
    {
        timedAction = Welding,
